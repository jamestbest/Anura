// pages
//  2916- opcode map (1 BYTE)
//  2910- opcode map key
//  527 - SIB information
//  2949- instr encoding
//  1257- MOV encoding
//  529 - REX prefix encoding
//  87  - SIB information (limited)
//  765 - CMOVcc encoding
//  521 - Instruction encoding
//  535 - VEX prefix encoding
//  558 - EVEX prefix encoding

// RULE ON NUMBERS
//  some places expect base 10, i.e. anywhere that isn't inside {} e.g. 4 bits
//  most places expect base 2 written as 0011 1011
//  numbers are written from high bit - low bit and will go until 8 bits are found
//  e.g. 1010 101 ow -- this is one BYTE as any non-alias identifier is 1 bit in size
//  e.g. 10 reg 001  -- this is one BYTE (assuming reg is a 3 bit ALIAS)
META= {
    .name= "x64"
}

// assume we're in long mode
//  so assume REX exists -- I don't want to do include protected mode
//   with stuff like single BYTE inc, just keep to long mode
//  SO is this --v needed?? -- I assume not
FLAG mode= 64bit | 32bit | 16bit default 64bit
FLAG opmode= 64bit | 32bit | 16bit | 8bit default 32bit
FLAG addrmode= 64bit | 32bit | 16bit default 64bit

VAR default_opmode OF opmode = 32bit

DATA ow 1 bit
DATA regx 3 BITS
DATA sibs 2 BITS

DATA imm8 1 BYTE
DATA imm16 2 BYTES
DATA imm32 4 BYTES
DATA imm64 8 BYTES

DATA disp8 1 BYTE
DATA disp16 2 BYTES
DATA disp32 4 BYTES

// REX.W is for OPERAND size
// REX.W = 64 Bit Operand Size
// REX.R is for Rrrr
// REX.R = Extension of the ModR/M reg field
// REX.X is for X(SIB.xxx)
// REX.X = Extension of the SIB index field
// REX.B is for B(SIB.bbb)
//       or for Brrr
//       or for B(RM.bbb)
// REX.B = Extension of the ModR/M r/m field, SIB base field, or Opcode reg field
DATA REX= {
  0100 .w=0 .r .x .b
}

DATA VEX= {
  // the two formats here are just to show how it gets parsed
  // the generated output combines these two together
  // which means fields that appear in both should have the same
  // default values
  1100 0100 .R .X .B .m(5) .W .v(4) .L .pp(2)
  1100 0101 .R .v(4) .L .pp(2)
}

ALIAS prefix= {
    REX
    VEX
    // EVEX
}

ALIAS lp1 1 BYTE= {
  0xF0= "LOCK"
  0xF2= "REPNE"  // REPNE,REPNZ,BND; BND has some restrictions to do with what the instr is
  0xF3= "REP"
}

ALIAS lp2 1 BYTE= {
  0x2E= "CS"
  0x36= "SS"
  0x3E= "DS"
  0x26= "ES"
  0x64= "FS"
  0x65= "GS"
}

ALIAS lp3 1 BYTE= {
  0x66= "OO"
}

ALIAS lp4 1 BYTE= {
  0x67= "AO"
}

ALIAS lprefix 1 BYTE= {
  lp1
  lp2
  lp3
  lp4
}

// some instructions access SPL etc. instead of AH etc.

ALIAS regT 3 BITS= {
  000= "AX", "AL", "AL", "EAX", "RAX"
  001= "CX", "CL", "AL", "ECX", "RCX"

  100= "SP", "SPL", "AL", "ESP", "RSP"
}

ALIAS regO= reg

  0000= "AX", "AL", "AL", "EAX", "RAX"
  0001= "CX", "CL", "CL", "ECX", "RCX"
  0010= "DX", "DL", "DL", "EDX", "RDX"
  0011= "BX", "BL", "BL", "EBX", "RBX"

RULE RIGHT ON regO, regT, reg {
  CHOOSE 0 if opmode == 16bit
  CHOOSE 1 if opmode == 8bit && REX.w != 1
  CHOOSE 2 if opmode == 8bit
  CHOOSE 3 if opmode == 32bit
  CHOOSE 4 if opmode == 64bit
}

RULE RIGHT ON regM {
  CHOOSE 0 if addrmode == 16bit
  CHOOSE 1 if 0
  CHOOSE 2 if 0
  CHOOSE 3 if addrmode == 32bit
  CHOOSE 4 if addrmode == 64bit
}

ALIAS cc 4 BITS= {
  0000= "O"
  0001= "NO"
  0010= "B"
  0011= "NB"
  0100= "E"
  0101= "NE"
  0110= "BE"
  0111= "NBE"
  1000= "S"
  1001= "NS"
  1010= "P"
  1011= "NP"
  1100= "L"
  1101= "NL"
  1110= "LE"
  1111= "NLE"
}

ALIAS regi 3 BITS = {
    regx = MAP regM REX.x regx
}
ALIAS regb 3 bits = {
    regx = MAP regM REX.b regx
}
ALIAS regop 3 bits = {
    regx = MAP regO REX.b regx
}
ALIAS regr 3 bits = {
    regx = MAP regO REX.r regx
}

DATA Mod 2 BITS

CALCULATE addrmode= {
    if lp4? then 32bit
    64bit
}

// this is a flat alias
// i.e. it has no l rules, so no reading from the stream
// they are special because they can appear in right rules
// without needing to be in the left side
FLAT ALIAS addr_ptr= {
    if opmode == 64bit then "QWORD PTR"
    if opmode == 32bit then "DWORD PTR"
    if opmode == 16bit then "WORD PTR"
    if opmode == 8bit then "BYTE PTR"
}

ALIAS sibsi 5 BITS= {
   sibs 100 = "0x0"
   sibs regi= "{regi} * {2 ^ sibs}"
}

ALIAS SIB_INT 1 BYTE= {
  sibsi 101 = {
    if Mod == 00 then {
        disp32 = "{sibsi} + {disp32}"
    }
    if Mod == 01 then {
        disp8 = "{sibsi} + {disp8} + [EBP]"
    }
    if Mod == 10 then {
        disp32 = "{sibsi} + {disp32} + [EBP]"
    }
  }
  sibsi regb = "{regb} + {sibsi}"
}

ALIAS SIB 1 BYTE= {
    SIB_INT = SIB_INT
}

ALIAS rm_INT= {
    if Mod != 11 then {
        100 SIB = SIB
    }

    if Mod == 00 then {
        101 disp32 = {
            when mode == 64bit then "rip + {disp32}"
            disp32
        }
        regb = regb
    }

    if Mod == 01 then {
        regb disp8 = "{regb} + {disp8}"
    }

    if Mod == 10 then {
        regb disp32 = "{regb} + {disp32}"
    }

    if Mod == 11 then {
        regop = regop
    }
}

ALIAS rm= {
    if Mod == 11 then {rm_INT=rm_INT}
    rm_INT = "[{rm_INT}]"
}

FLAT ALIAS rm_ptr= {
    if Mod == 11 then rm
    if 1 then "{addr_ptr} {rm}"
}

ALIAS ModRM= {
  Mod regr rm= "{regr}, {rm}"
}

CALCULATE opmode= {
  if ow then 8bit
  if REX.w then 64bit
  if lp3? then 16bit
  default_opmode
}

ALIAS imm= {
  if opmode == 8bit then {imm8=imm8}
  if opmode == 16bit then {imm16=imm16}
  if opmode == 32bit then {imm32=imm32}
  if opmode == 64bit then {imm64=imm64}
}

ALIAS immM32= {
  if opmode == 8bit then {imm8=imm8}
  if opmode == 16bit then {imm16=imm16}
  if opmode == 32bit then {imm32=imm32}
  if opmode == 64bit then {imm32=imm32}
}

// opcodes are matched up to the next non-0|1
//  i.e. each level is expanded to it's level of 1|0s
//  and each is an opcode

// Mod switch used to switch the reg and rm of ModRM in output
DATA ms 1 BIT

FLAT ALIAS ModRMS= {
    if ms == 0 then "{rm_ptr}, {regr}"
    if ms == 1 then "{regr}, {rm}"
}

ALIAS MOV= {
  1000 10 ms ~ow ModRM= "MOV {ModRMS}"
  1100 011 ~ow Mod 000 rm immM32 = "MOV {rm_ptr}, {immM32}"
  1011 ~ow regO imm= "MOV {regT}, {imm}" //todo fix this regT
}

// ALIAS regA= "RAX"
FLAT ALIAS regA = {
    if 1 then MAP reg 0000
}

ALIAS ADC= {
  0000 111 ow immM32= "ADC RAX, {immM32}" // Add with carry to regA immM32 ## regA= regA + immM32 + carrybit
  1000 000 ow immM32= "ADC {reg}, {immM32}"
  1000 0011 Mod 010 regO imm8= "ADC {reg}, {imm8}"
  0001 00 ms ow ModRM = "ADC {ModRMS}"
}

ALIAS PUSH= {
  WITH default_opmode = 64bit {
    0xFF Mod 110 rm = "PUSH {rm}"
    0101 0 regop = "PUSH {regop}"
  }
  0110 10 ow 0 immM32 = "PUSH {immM32}"
}

ALIAS CMOV= {
  0100 cc ModRM = "CMOV{cc} {ModRM}"
}

ALIAS ModRM_5= {
    Mod 101 rm= rm
}

ALIAS ModRM_2= {
    Mod 010 rm= rm
}

ALIAS SUB= {
  0010 110 ~ow immM32 = "SUB {regA}, {immM32}"
  1000 000 ~ow ModRM_5 immM32 = "SUB {ModRM_5}, {immM32}"
  1000 0011 ModRM_5 imm8 = "SUB {ModRM_5}, {imm8}"
}

ALIAS op2= {
  // 0xE6 ModRM= "CVTDQ2PD {xyzreg}, {xyzrm}"  // change on E/VEX //  REQUIRE REP
  0x1E 0xFA = "ENDBR64"
  CMOV
}

ALIAS op= {
  0x0f op2= op2
  MOV
  // ADC
  PUSH
  SUB
}

STRUCTURE lprefix* prefix? op = op
